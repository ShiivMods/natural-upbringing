namespace = nu_assimilation


# =============================================================
# CULTURAL ASSIMILATION
# =============================================================

nu_assimilation.0001 = {
	type = character_event
	theme = family

	immediate = {
		nu_prepare_assimilation_event_context_effect = yes
		set_local_variable = { name = nu_assimilation_event_kind value = flag:culture }
		nu_prepare_cultural_assimilation_reaction_effect = yes
		nu_select_cultural_assimilation_reaction_effect = yes
	}

	title = nu_assimilation.0001.t

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:nu_assimilated_child = {
						has_character_flag = nu_recent_cultural_assimilation_rapid
					}
				}

				desc = nu_assimilation.0001.desc_rapid
			}

			desc = nu_assimilation.0001.desc
		}

		first_valid = {

			triggered_desc = {
				trigger = {
					exists = scope:nu_cultural_assimilation_guardian
					exists = scope:nu_cultural_assimilation_court_owner

					scope:nu_assimilated_child = {
						has_character_flag = nu_last_cultural_assimilation_had_servants
					}
				}

				desc = nu_assimilation.0001.context_guardian_court_servants
			}

			triggered_desc = {
				trigger = {
					exists = scope:nu_cultural_assimilation_guardian
					exists = scope:nu_cultural_assimilation_court_owner
				}

				desc = nu_assimilation.0001.context_guardian_court
			}

			triggered_desc = {
				trigger = {
					exists = scope:nu_cultural_assimilation_court_owner
				}

				desc = nu_assimilation.0001.context_court
			}

			triggered_desc = {
				trigger = {
					exists = scope:nu_cultural_assimilation_guardian
				}

				desc = nu_assimilation.0001.context_guardian
			}
		}
	}

	left_portrait = {
		character = scope:nu_assimilated_child
	}

	# =========================================================
	# PERSONALITY REACTIONS - ALL VANILLA PERSONALITY TRAITS
	# =========================================================
	# Each option is independent. Characters with four relevant
	# personality traits (e.g. in total conversions such as AGOT)
	# can therefore receive four distinct reactions.
	# =========================================================


	option = {
		name = nu_assimilation.0001.rejection

		trigger = {
			local_var:nu_assimilation_reaction = flag:rejection
		}
		nu_apply_assimilation_rejection_consequence_effect = yes
	}

	option = {
		name = nu_assimilation.0001.mistrust

		trigger = {
			local_var:nu_assimilation_reaction = flag:mistrust
		}
		nu_apply_assimilation_mistrust_consequence_effect = yes
	}

	option = {
		name = nu_assimilation.0001.acceptance

		trigger = {
			local_var:nu_assimilation_reaction = flag:acceptance
		}
		nu_apply_assimilation_acceptance_consequence_effect = yes
	}

	option = {
		name = nu_assimilation.0001.reflection

		trigger = {
			local_var:nu_assimilation_reaction = flag:reflection
		}
		nu_apply_assimilation_reflection_consequence_effect = yes
	}

	option = {
		name = nu_assimilation.0001.pragmatism

		trigger = {
			local_var:nu_assimilation_reaction = flag:pragmatism
		}
	}

	option = {
		name = nu_assimilation.0001.adaptation

		trigger = {
			local_var:nu_assimilation_reaction = flag:adaptation
		}
	}

	option = {
		name = nu_assimilation.0001.fallback

		trigger = {
			local_var:nu_assimilation_reaction = flag:fallback
		}
	}
}
