	type = letter_event
	opening = nu_foreign_education.0001.opening
	desc = {

			}
			triggered_desc = {
				trigger = {

						}
					}
				}
				desc = nu_foreign_education.0001.desc.reserved
			}
			triggered_desc = {
				trigger = {
					scope:nu_foreign_education_child = {
						OR = {
