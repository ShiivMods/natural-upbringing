namespace = nu_debug


nu_debug.0001 = {
	type = character_event
	theme = education

	title = nu_debug.0001.t

	immediate = {

		# Rebuild the inspected character scope inside the event.
		if = {
			limit = {
				exists = var:nu_debug_inspection_target
			}

			var:nu_debug_inspection_target = {
				save_scope_as = nu_debug_child
			}

			remove_variable = nu_debug_inspection_target
		}

		# Resolve all three exposure tracks independently.
		if = {
			limit = {
				exists = scope:nu_debug_child
			}

			scope:nu_debug_child = {

				if = {
					limit = {
						exists = var:nu_culture_exposure_target
					}

					var:nu_culture_exposure_target = {
						save_scope_as = nu_debug_culture_target
					}
				}

				if = {
					limit = {
						exists = var:nu_belief_exposure_target
					}

					var:nu_belief_exposure_target = {
						save_scope_as = nu_debug_faith_target
					}
				}

				if = {
					limit = {
						exists = var:nu_rite_exposure_target
					}

					var:nu_rite_exposure_target = {
						save_scope_as = nu_debug_rite_target
					}
				}
			}
		}
	}

	desc = {
		desc = nu_debug.0001.desc

		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:nu_debug_culture_target
				}
				desc = nu_debug.0001.culture
			}
			desc = nu_debug.0001.culture_none
		}

		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:nu_debug_faith_target
				}
				desc = nu_debug.0001.faith
			}
			desc = nu_debug.0001.faith_none
		}

		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:nu_debug_rite_target
				}
				desc = nu_debug.0001.rite
			}
			desc = nu_debug.0001.rite_none
		}

		first_valid = {
			triggered_desc = {
				trigger = {
					scope:nu_debug_child = {
						has_character_flag = nu_homesickness_active
					}
				}
				desc = nu_debug.0001.homesickness_active
			}
			desc = nu_debug.0001.homesickness_inactive
		}
	}

	left_portrait = {
		character = scope:nu_debug_child
	}

	option = {
		name = nu_debug.0001.a
	}
}


nu_debug.0002 = {
	type = character_event
	theme = education
	title = nu_debug.0002.t

	immediate = {
		if = {
			limit = { exists = var:nu_debug_conversion_target }
			var:nu_debug_conversion_target = { save_scope_as = nu_debug_conversion_child }
			remove_variable = nu_debug_conversion_target
		}
		if = {
			limit = { exists = scope:nu_debug_conversion_child }
			scope:nu_debug_conversion_child = {
				if = {
					limit = { exists = var:nu_culture_exposure_target }
					var:nu_culture_exposure_target = { save_scope_as = nu_debug_conversion_culture }
				}
				if = {
					limit = { exists = var:nu_belief_exposure_target }
					var:nu_belief_exposure_target = { save_scope_as = nu_debug_conversion_faith }
				}
				if = {
					limit = { exists = var:nu_rite_exposure_target }
					var:nu_rite_exposure_target = { save_scope_as = nu_debug_conversion_rite }
				}
			}
		}
	}

	desc = nu_debug.0002.desc
	left_portrait = { character = scope:nu_debug_conversion_child }

	option = {
		name = nu_debug.0002.culture
		trigger = { exists = scope:nu_debug_conversion_culture }
		scope:nu_debug_conversion_child = {
			save_scope_as = nu_assimilated_child
			nu_debug_force_cultural_assimilation_effect = yes
		}
		trigger_event = nu_assimilation.0001
		scope:nu_debug_conversion_child = {
			nu_notify_player_guardian_of_assimilation_effect = yes
			remove_character_flag = nu_culture_assimilated_this_birthday
			remove_character_flag = nu_assimilation_happened_this_birthday
		}
	}

	option = {
		name = nu_debug.0002.faith
		trigger = { exists = scope:nu_debug_conversion_faith }
		scope:nu_debug_conversion_child = {
			save_scope_as = nu_assimilated_child
			nu_debug_force_belief_assimilation_effect = yes
		}
		trigger_event = nu_assimilation.0002
		scope:nu_debug_conversion_child = {
			nu_notify_player_guardian_of_belief_assimilation_effect = yes
			remove_character_flag = nu_belief_assimilated_this_birthday
			remove_character_flag = nu_assimilation_happened_this_birthday
		}
	}

	option = {
		name = nu_debug.0002.rite
		trigger = { exists = scope:nu_debug_conversion_rite }
		scope:nu_debug_conversion_child = {
			save_scope_as = nu_assimilated_child
			nu_debug_force_religious_assimilation_effect = yes
		}
		trigger_event = nu_assimilation.0002
		scope:nu_debug_conversion_child = {
			nu_notify_player_guardian_of_assimilation_effect = yes
			remove_character_flag = nu_faith_assimilated_this_birthday
			remove_character_flag = nu_assimilation_happened_this_birthday
		}
	}

	option = {
		name = nu_debug.0002.all
		trigger = {
			exists = scope:nu_debug_conversion_culture
			OR = {
				exists = scope:nu_debug_conversion_faith
				exists = scope:nu_debug_conversion_rite
			}
		}
		scope:nu_debug_conversion_child = {
			save_scope_as = nu_assimilated_child
			nu_debug_force_cultural_assimilation_effect = yes
			if = {
				limit = { exists = var:nu_belief_exposure_target }
				nu_debug_force_belief_assimilation_effect = yes
			}
			else_if = {
				limit = { exists = var:nu_rite_exposure_target }
				nu_debug_force_religious_assimilation_effect = yes
			}
		}
		trigger_event = nu_assimilation.0003
		scope:nu_debug_conversion_child = {
			if = {
				limit = { exists = scope:nu_debug_conversion_faith }
				nu_notify_player_guardian_of_belief_assimilation_effect = yes
			}
			else = {
				nu_notify_player_guardian_of_assimilation_effect = yes
			}
			remove_character_flag = nu_culture_assimilated_this_birthday
			remove_character_flag = nu_faith_assimilated_this_birthday
			remove_character_flag = nu_belief_assimilated_this_birthday
			remove_character_flag = nu_assimilation_happened_this_birthday
		}
	}

	option = { name = nu_debug.0002.cancel }
}

nu_debug.0003 = {
	type = character_event
	theme = education

	title = nu_debug.0003.t
	desc = nu_debug.0003.desc

	immediate = {
		if = {
			limit = { exists = var:nu_debug_education_event_target }
			var:nu_debug_education_event_target = {
				save_scope_as = nu_debug_education_child
			}
			remove_variable = nu_debug_education_event_target
		}

		if = {
			limit = { exists = scope:nu_debug_education_child }

			scope:nu_debug_education_child = {
				save_scope_as = nu_foreign_education_child

				random_relation = {
					type = guardian
					save_scope_as = nu_foreign_education_guardian
				}

				court_owner = {
					save_scope_as = nu_foreign_education_current_court
					culture = { save_scope_as = nu_foreign_education_current_culture }
					faith = { save_scope_as = nu_foreign_education_current_faith }
					rite = { save_scope_as = nu_foreign_education_current_rite }
				}
			}

			if = {
				limit = { exists = scope:nu_foreign_education_current_court }
				scope:nu_foreign_education_current_court = {
					if = {
						limit = {
							any_courtier_or_guest = {
								age >= 6
								is_adult = no
								this != scope:nu_debug_education_child
							}
						}
						random_courtier_or_guest = {
							limit = {
								age >= 6
								is_adult = no
								this != scope:nu_debug_education_child
							}
							save_scope_as = nu_debug_peer
						}
					}
					if = {
						limit = {
							any_courtier_or_guest = {
								is_adult = yes
								this != scope:nu_debug_education_child
							}
						}
						random_courtier_or_guest = {
							limit = {
								is_adult = yes
								this != scope:nu_debug_education_child
							}
							save_scope_as = nu_debug_adult_target
						}
					}
				}
			}

			scope:nu_debug_education_child = {
				if = {
					limit = {
						any_relation = {
							type = rival
							court_owner ?= {
								this = scope:nu_foreign_education_current_court
							}
						}
					}
					random_relation = {
						type = rival
						limit = {
							court_owner ?= {
								this = scope:nu_foreign_education_current_court
							}
						}
						save_scope_as = nu_debug_rival
					}
				}
			}
		}
	}

	left_portrait = {
		character = scope:nu_debug_education_child
	}

	option = {
		name = nu_debug.0003.report
		hidden_effect = {
			scope:nu_debug_education_child = {
				nu_resolve_general_adaptation_report_effect = yes
			}
		}
		trigger_event = nu_foreign_education.0001
	}

	option = {
		name = nu_debug.0003.friendship
		trigger = { exists = scope:nu_debug_peer }
		hidden_effect = {
			scope:nu_debug_peer = { save_scope_as = nu_foreign_education_peer }
			scope:nu_debug_education_child = {
				if = {
					limit = {
						can_set_relation_friend_trigger = {
							CHARACTER = scope:nu_foreign_education_peer
						}
					}
					set_relation_friend = {
						reason = friend_childhood_random
						target = scope:nu_foreign_education_peer
					}
				}
				nu_resolve_unexpected_friendship_effect = yes
			}
		}
		if = {
			limit = {
				scope:nu_debug_education_child = {
					has_character_flag = nu_foreign_crush_from_friendship
					has_character_flag = nu_foreign_crush_reportable
				}
			}
			trigger_event = nu_foreign_education.0070
		}
		else = {
			trigger_event = nu_foreign_education.0010
		}
	}

	option = {
		name = nu_debug.0003.foreigner
		hidden_effect = {
			if = {
				limit = { exists = scope:nu_debug_peer }
				scope:nu_debug_peer = { save_scope_as = nu_foreign_education_peer }
			}
			scope:nu_debug_education_child = {
				nu_resolve_foreigner_at_court_effect = yes
			}
		}
		trigger_event = nu_foreign_education.0020
	}

	option = {
		name = nu_debug.0003.custom
		hidden_effect = {
			scope:nu_debug_education_child = {
				nu_resolve_strange_custom_effect = yes
			}
		}
		trigger_event = nu_foreign_education.0030
	}

	option = {
		name = nu_debug.0003.words
		trigger_event = nu_foreign_education.0040
	}

	option = {
		name = nu_debug.0003.two_worlds
		trigger_event = nu_foreign_education.0050
	}

	option = {
		name = nu_debug.0003.crush
		if = {
			limit = {
				scope:nu_foreign_education_current_court = {
					any_courtier_or_guest = {
						this != scope:nu_debug_education_child
						scope:nu_debug_education_child = {
							nu_can_form_foreign_crush_with_trigger = { CHARACTER = prev }
							nu_foreign_crush_is_reportable_trigger = { CHARACTER = prev }
						}
					}
				}
			}
			hidden_effect = {
				scope:nu_foreign_education_current_court = {
					random_courtier_or_guest = {
						limit = {
							this != scope:nu_debug_education_child
							scope:nu_debug_education_child = {
								nu_can_form_foreign_crush_with_trigger = { CHARACTER = prev }
								nu_foreign_crush_is_reportable_trigger = { CHARACTER = prev }
							}
						}
						save_scope_as = nu_foreign_education_peer
					}
				}
				scope:nu_debug_education_child = { nu_make_foreign_crush_effect = yes }
			}
			trigger_event = nu_foreign_education.0070
		}
		else = { trigger_event = nu_debug.0007 }
	}

	option = {
		name = nu_debug.0003.rival
		if = {
			limit = { exists = scope:nu_debug_rival }
			scope:nu_debug_rival = { save_scope_as = nu_foreign_education_rival }
		}
		else_if = {
			limit = {
				exists = scope:nu_debug_peer
				scope:nu_debug_education_child = {
					can_set_relation_rival_even_if_teen_trigger = { CHARACTER = scope:nu_debug_peer }
				}
			}
			scope:nu_debug_education_child = { set_relation_rival = scope:nu_debug_peer }
			scope:nu_debug_peer = { save_scope_as = nu_foreign_education_rival }
		}
		if = {
			limit = { exists = scope:nu_foreign_education_rival }
			hidden_effect = {
				scope:nu_debug_education_child = { nu_resolve_rival_news_effect = yes }
			}
			if = {
				limit = {
					scope:nu_debug_education_child = {
						has_character_flag = nu_rival_news_crush
						has_character_flag = nu_foreign_crush_reportable
					}
				}
				scope:nu_foreign_education_rival = { save_scope_as = nu_foreign_education_peer }
				trigger_event = nu_foreign_education.0070
			}
			else = { trigger_event = nu_foreign_education.0080 }
		}
		else = { trigger_event = nu_debug.0008 }
	}

	option = {
		name = nu_debug.0003.focus
		trigger_event = nu_debug.0004
	}

	option = {
		name = nu_debug.0003.cancel
	}
}

# =============================================================
# 0010 - FORCE RUNAWAY WITH CHILD AS ROOT
# =============================================================

nu_debug.0010 = {
	type = character_event
	hidden = yes

	immediate = {
		# Give the child the minimum persistent homesickness state
		# required by the real runaway arc.
		if = {
			limit = {
				NOT = { has_character_flag = nu_homesickness_active }
			}
			add_character_flag = nu_homesickness_active
			set_variable = {
				name = nu_homesickness_active_years
				value = 1
			}
			set_variable = {
				name = nu_homesickness_severity
				value = 1
			}
		}
		else_if = {
			limit = {
				OR = {
					NOT = { has_variable = nu_homesickness_active_years }
					var:nu_homesickness_active_years < 1
				}
			}
			set_variable = {
				name = nu_homesickness_active_years
				value = 1
			}
		}

		nu_start_runaway_effect = yes
	}
}
