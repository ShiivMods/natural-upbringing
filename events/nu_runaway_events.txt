# =============================================================
# 0100 - SUCCESSFUL RETURN, PARENT DECISION
# =============================================================

nu_runaway.0100 = {
	type = character_event
	theme = family

	title = nu_runaway.0100.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:nu_runaway_child = {
						has_character_flag = nu_runaway_result_critical_success
					}
				}
				desc = nu_runaway.0100.desc.critical
			}
			triggered_desc = {
				trigger = {
					scope:nu_runaway_child = {
						has_character_flag = nu_runaway_result_success
					}
				}
				desc = nu_runaway.0100.desc.success
			}
			triggered_desc = {
				trigger = {
					scope:nu_runaway_child = {
						has_character_flag = nu_runaway_result_helped
					}
				}
				desc = nu_runaway.0100.desc.helped
			}
			desc = nu_runaway.0100.desc.escorted
		}
	}

	left_portrait = {
		character = scope:nu_runaway_child
	}

	right_portrait = {
		character = scope:nu_runaway_guardian
	}

	option = {
		name = nu_runaway.0100.a

		show_as_tooltip = {
			scope:nu_runaway_child = {
				if = { limit = { has_character_flag = nu_runaway_result_critical_success } add_prestige = medium_prestige_gain }
				else = { add_prestige = minor_prestige_gain }
			}
		}

		if = {
			limit = {
				exists = scope:nu_runaway_child
				exists = scope:nu_runaway_guardian
				scope:nu_runaway_child = {
					is_alive = yes
					any_relation = {
						type = guardian
						this = scope:nu_runaway_guardian
					}
				}
			}

			scope:nu_runaway_child = {
				add_opinion = {
					target = root
					modifier = nu_runaway_parent_accepted_return
				}
			}

			# Preserve the parent explicitly before changing character scope.
			save_scope_as = nu_runaway_return_parent

			remove_guardian_effect = {
				GUARDIAN = scope:nu_runaway_guardian
				WARD = scope:nu_runaway_child
				RETURN_WARD = no
				HIDE_OPINION = yes
			}

			# The story has already established that the child reached the
			# parent. Vanilla court transfers pair set_employer with
			# return_to_court so the character actually joins the court.
			scope:nu_runaway_child = {
				if = {
					limit = { is_ruler = no }
					set_employer = scope:nu_runaway_return_parent
					return_to_court = yes
				}
				else = {
					move_to_pool_at = scope:nu_runaway_return_parent.location
				}
			}

			scope:nu_runaway_child = {
				remove_variable = nu_homesickness_abroad_years
				nu_clear_homesickness_effect = yes
			}
		}

		stress_impact = {
			compassionate = medium_stress_impact_loss
			callous = minor_stress_impact_gain
			stubborn = minor_stress_impact_gain
		}

		ai_chance = {
			base = 70
			modifier = { add = 50 has_trait = compassionate }
			modifier = { add = 20 has_trait = trusting }
			modifier = { add = -40 has_trait = callous }
			modifier = { add = -25 has_trait = stubborn }
			modifier = { add = -15 has_trait = ambitious }
		}
	}

	option = {
		name = nu_runaway.0100.b

		show_as_tooltip = {
			scope:nu_runaway_child = {
				if = { limit = { has_character_flag = nu_runaway_result_critical_success } add_prestige = medium_prestige_gain }
				else = { add_prestige = minor_prestige_gain }
			}
		}

		if = {
			limit = {
				exists = scope:nu_runaway_child
				scope:nu_runaway_child = {
					is_alive = yes
				}
			}

			scope:nu_runaway_child = {
				add_stress = medium_stress_gain

				add_opinion = {
					target = root
					modifier = nu_runaway_parent_sent_back
				}

				if = {
					limit = { has_variable = nu_homesickness_severity }
					change_variable = {
						name = nu_homesickness_severity
						add = 2
					}
				}
				else = {
					set_variable = {
						name = nu_homesickness_severity
						value = 3
					}
				}

				if = {
					limit = {
						NOT = { has_character_flag = nu_homesickness_active }
					}
					add_character_flag = nu_homesickness_active
				}

				if = {
					limit = {
						NOT = { has_variable = nu_homesickness_active_years }
					}
					set_variable = {
						name = nu_homesickness_active_years
						value = 1
					}
				}
			}
		}

		stress_impact = {
			compassionate = medium_stress_impact_gain
			callous = miniscule_stress_impact_loss
			stubborn = minor_stress_impact_loss
			ambitious = miniscule_stress_impact_loss
		}

		ai_chance = {
			base = 30
			modifier = { add = 50 has_trait = callous }
			modifier = { add = 30 has_trait = stubborn }
			modifier = { add = 20 has_trait = ambitious }
			modifier = { add = -30 has_trait = compassionate }
			modifier = { add = -10 has_trait = trusting }
		}
	}
}

# 9000 - HIDDEN MONTHLY PROGRESSION
# =============================================================

nu_runaway.9000 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_character_flag = nu_runaway_active
				is_alive = yes
			}

			nu_runaway_monthly_progress_effect = yes

			if = {
				limit = {
					has_character_flag = nu_runaway_active
					is_alive = yes
				}
				trigger_event = {
					id = nu_runaway.9000
					days = 30
				}
			}
		}
	}
}
