nu_add_focus_education_points_effect = {
	if = {
		limit = { has_focus = education_diplomacy }
		if = {
			limit = { NOT = { has_variable = education_diplomacy_variable } }
			set_variable = { name = education_diplomacy_variable value = 0 }
		}
		change_variable = { name = education_diplomacy_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_martial }
		if = {
			limit = { NOT = { has_variable = education_martial_variable } }
			set_variable = { name = education_martial_variable value = 0 }
		}
		change_variable = { name = education_martial_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_stewardship }
		if = {
			limit = { NOT = { has_variable = education_stewardship_variable } }
			set_variable = { name = education_stewardship_variable value = 0 }
		}
		change_variable = { name = education_stewardship_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_intrigue }
		if = {
			limit = { NOT = { has_variable = education_intrigue_variable } }
			set_variable = { name = education_intrigue_variable value = 0 }
		}
		change_variable = { name = education_intrigue_variable add = $VALUE$ }
	}
	else_if = {
		limit = { has_focus = education_learning }
		if = {
			limit = { NOT = { has_variable = education_learning_variable } }
			set_variable = { name = education_learning_variable value = 0 }
		}
		change_variable = { name = education_learning_variable add = $VALUE$ }
	}
}


nu_clear_focus_outcome_flags_effect = {
	remove_character_flag = nu_focus_outcome_strong
	remove_character_flag = nu_focus_outcome_mixed
	remove_character_flag = nu_focus_outcome_bad
	remove_character_flag = nu_focus_outcome_restrained
	remove_character_flag = nu_focus_outcome_forceful
	remove_character_flag = nu_focus_outcome_sadistic
	remove_character_flag = nu_focus_outcome_local
	remove_character_flag = nu_focus_outcome_reject
}


nu_set_focus_outcome_flag_effect = {
	add_character_flag = {
		flag = $FLAG$
		days = 7
	}
}


# Diplomacy 1: Between Two Quarrels
nu_resolve_focus_diplomacy_quarrels_effect = {
	nu_clear_focus_outcome_flags_effect = yes

	if = {
		limit = {
			OR = {
				has_trait = gregarious
				has_trait = patient
				has_trait = compassionate
				has_trait = just
				diplomacy >= 12
			}
		}
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_strong }
		nu_add_focus_education_points_effect = { VALUE = 2 }

		if = {
			limit = { exists = scope:nu_focus_target }
			random = {
				chance = 10
				add_opinion = {
					target = scope:nu_focus_target
					modifier = nu_foreign_education_social_opinion
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = wrathful
				has_trait = arrogant
				has_trait = arbitrary
				has_trait = impatient
				diplomacy <= 5
			}
		}
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_bad }
	}
	else = {
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_mixed }
		nu_add_focus_education_points_effect = { VALUE = 1 }
	}
}

# Martial 2: One Blow Too Many

		if = {
			limit = {
				exists = scope:nu_focus_peer
				can_set_relation_rival_even_if_teen_trigger = { CHARACTER = scope:nu_focus_peer }

# Intrigue 1: Not Meant for Their Ears
nu_resolve_focus_intrigue_overheard_effect = {
	nu_clear_focus_outcome_flags_effect = yes

	if = {
		limit = {
			OR = {
				has_trait = deceitful
				has_trait = paranoid
				has_trait = calm
				intrigue >= 12
			}
		}
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_strong }
		nu_add_focus_education_points_effect = { VALUE = 2 }

		if = {
			limit = {
				exists = scope:nu_focus_target
				can_add_hook = {
					target = scope:nu_focus_target
					type = favor_hook
				}
			}
			random = {
				chance = 12
				add_hook = {
					target = scope:nu_focus_target
					type = favor_hook
				}
			}
		}
	}
	else_if = {
		limit = {
			OR = {
				has_trait = honest
				has_trait = trusting
				has_trait = compassionate
			}
		}
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_mixed }
		nu_add_focus_education_points_effect = { VALUE = 1 }

		if = {
			limit = {
				exists = scope:nu_focus_target
				exists = scope:nu_foreign_education_guardian
				scope:nu_foreign_education_guardian = {
					can_add_hook = {
						target = scope:nu_focus_target
						type = favor_hook
					}
				}
			}
			random = {
				chance = 8
				scope:nu_foreign_education_guardian = {
					add_hook = {
						target = scope:nu_focus_target
						type = favor_hook
					}
				}
			}
		}
	}
	else_if = {
		limit = { intrigue <= 5 }
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_bad }
		add_stress = minor_stress_gain
	}
	else = {
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_mixed }
		nu_add_focus_education_points_effect = { VALUE = 1 }
	}
}

# Learning 1: Two Teachings, Two Truths

		limit = {
			OR = {

			}
		}
		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_strong }

		nu_add_focus_education_points_effect = { VALUE = 1 }

		nu_set_focus_outcome_flag_effect = { FLAG = nu_focus_outcome_local }
