nu_update_exposure_tracking_effect = {

	# ---------------------------------------------------------
	# Find the child's guardian, if one exists.
	# A child can only have one guardian in normal vanilla use.
	# ---------------------------------------------------------

	if = {
		limit = {
			any_relation = {
				type = guardian
				always = yes
			}
		}

		random_relation = {
			type = guardian
			save_temporary_scope_as = nu_guardian
		}
	}


	# =========================================================
	# CULTURAL EXPOSURE - 1.2.1 MULTI-TRACK ENGINE
	# =========================================================

	# Culture is now tracked from birth and can have several simultaneous
	# targets. This effect rebuilds the legacy single-target variables as a
	# compatibility view for older 1.2 events and debug tools.
	nu_update_multi_cultural_exposure_effect = yes


	# =========================================================
	# RELIGIOUS EXPOSURE
	# =========================================================

	# Compatibility migration from Natural Upbringing 1.1.
	# The old target stored a faith scope; the current game uses
	# rites for religious education, so restart that exposure safely.
	if = {
		limit = { exists = var:nu_faith_exposure_target }
		remove_variable = nu_faith_exposure_target
		remove_variable = nu_faith_exposure_years
		remove_variable = nu_faith_exposure_score
		remove_variable = nu_faith_exposure_floor
	}

	# Resolve one religious environment only.
	# Guardian influence has priority over the surrounding court.
	# Once that source is selected, faith is evaluated before rite:
	# - different faith -> faith exposure only;
	# - same faith, different rite -> rite exposure only.
	if = {
		limit = {
			exists = scope:nu_guardian
			scope:nu_guardian.faith != root.faith
		}

		scope:nu_guardian.faith = {
			save_temporary_scope_as = nu_belief_environment
		}
	}

	else_if = {
		limit = {
			exists = scope:nu_guardian
			scope:nu_guardian.faith = root.faith
			scope:nu_guardian.rite != root.rite
		}

		scope:nu_guardian.rite = {
			save_temporary_scope_as = nu_faith_environment
		}
	}

	else_if = {
		limit = {
			court_owner ?= {
				faith != root.faith
			}
		}

		court_owner.faith = {
			save_temporary_scope_as = nu_belief_environment
		}
	}

	else_if = {
		limit = {
			court_owner ?= {
				faith = root.faith
				rite != root.rite
			}
		}

		court_owner.rite = {
			save_temporary_scope_as = nu_faith_environment
		}
	}


	# ---------------------------------------------------------
	# We are currently exposed to a foreign rite within our own faith.
	# ---------------------------------------------------------

	if = {
		limit = {
			exists = scope:nu_faith_environment
		}

		# Same rite as last year:
		# continue accumulating exposure.
		if = {
			limit = {
				exists = var:nu_rite_exposure_target
				var:nu_rite_exposure_target = scope:nu_faith_environment
			}

			if = {
				limit = {
					NOT = {
						has_variable = nu_faith_exposure_years
					}
				}

				set_variable = {
					name = nu_faith_exposure_years
					value = 0
				}
			}

			if = {
				limit = {
					NOT = {
						has_variable = nu_faith_exposure_score
					}
				}

				set_variable = {
					name = nu_faith_exposure_score
					value = 0
				}
			}

			change_variable = {
				name = nu_faith_exposure_years
				add = 1
			}

			change_variable = {
				name = nu_faith_exposure_score
				add = nu_religious_exposure_yearly_gain
			}

			if = {
				limit = {
					var:nu_faith_exposure_score >= 50
				}

				set_variable = {
					name = nu_faith_exposure_floor
					value = 50
				}
			}

			else_if = {
				limit = {
					var:nu_faith_exposure_score >= 30

					OR = {
						NOT = {
							has_variable = nu_faith_exposure_floor
						}

						var:nu_faith_exposure_floor < 30
					}
				}

				set_variable = {
					name = nu_faith_exposure_floor
					value = 30
				}
			}
		}

		# New rite environment:
		# begin tracking the new rite from scratch.
		else = {
			set_variable = {
				name = nu_rite_exposure_target
				value = scope:nu_faith_environment
			}

			set_variable = {
				name = nu_faith_exposure_years
				value = 1
			}

			set_variable = {
				name = nu_faith_exposure_score
				value = nu_religious_exposure_yearly_gain
			}

			remove_variable = nu_faith_exposure_floor
		}
	}

	# ---------------------------------------------------------
	# The child has adopted the rite previously tracked.
	# The old exposure is no longer relevant.
	# ---------------------------------------------------------

	else_if = {
		limit = {
			exists = var:nu_rite_exposure_target
			var:nu_rite_exposure_target = root.rite
		}

		remove_variable = nu_rite_exposure_target
		remove_variable = nu_faith_exposure_years
		remove_variable = nu_faith_exposure_score
		remove_variable = nu_faith_exposure_floor
	}

	# ---------------------------------------------------------
	# No current foreign religious influence.
	# Previous exposure fades gradually.
	# ---------------------------------------------------------

	else_if = {
		limit = {
			has_variable = nu_faith_exposure_score
		}

		change_variable = {
			name = nu_faith_exposure_score
			add = -5
		}

		if = {
			limit = {
				has_variable = nu_faith_exposure_floor
				var:nu_faith_exposure_floor >= 50
				var:nu_faith_exposure_score < 50
			}

			set_variable = {
				name = nu_faith_exposure_score
				value = 50
			}
		}

		else_if = {
			limit = {
				has_variable = nu_faith_exposure_floor
				var:nu_faith_exposure_floor >= 30
				var:nu_faith_exposure_score < 30
			}

			set_variable = {
				name = nu_faith_exposure_score
				value = 30
			}
		}

		else_if = {
			limit = {
				var:nu_faith_exposure_score <= 0
			}

			remove_variable = nu_rite_exposure_target
			remove_variable = nu_faith_exposure_years
			remove_variable = nu_faith_exposure_score
			remove_variable = nu_faith_exposure_floor
		}
	}

	# =========================================================
	# FAITH EXPOSURE (separate from the legacy RITE exposure)
	# =========================================================
	# Keep rite variables for compatibility with existing 1.2 saves.

	# ---------------------------------------------------------
	# We are currently exposed to a foreign faith.
	# ---------------------------------------------------------

	if = {
		limit = {
			exists = scope:nu_belief_environment
		}

		# Same faith as last year:
		# continue accumulating exposure.
		if = {
			limit = {
				exists = var:nu_belief_exposure_target
				var:nu_belief_exposure_target = scope:nu_belief_environment
			}

			if = {
				limit = {
					NOT = {
						has_variable = nu_belief_exposure_years
					}
				}

				set_variable = {
					name = nu_belief_exposure_years
					value = 0
				}
			}

			if = {
				limit = {
					NOT = {
						has_variable = nu_belief_exposure_score
					}
				}

				set_variable = {
					name = nu_belief_exposure_score
					value = 0
				}
			}

			change_variable = {
				name = nu_belief_exposure_years
				add = 1
			}

			change_variable = {
				name = nu_belief_exposure_score
				add = nu_belief_exposure_yearly_gain
			}

			if = {
				limit = {
					var:nu_belief_exposure_score >= 50
				}

				set_variable = {
					name = nu_belief_exposure_floor
					value = 50
				}
			}

			else_if = {
				limit = {
					var:nu_belief_exposure_score >= 30

					OR = {
						NOT = {
							has_variable = nu_belief_exposure_floor
						}

						var:nu_belief_exposure_floor < 30
					}
				}

				set_variable = {
					name = nu_belief_exposure_floor
					value = 30
				}
			}
		}

		# New religious environment:
		# begin tracking the new faith from scratch.
		else = {
			set_variable = {
				name = nu_belief_exposure_target
				value = scope:nu_belief_environment
			}

			set_variable = {
				name = nu_belief_exposure_years
				value = 1
			}

			set_variable = {
				name = nu_belief_exposure_score
				value = nu_belief_exposure_yearly_gain
			}

			remove_variable = nu_belief_exposure_floor
		}
	}

	# ---------------------------------------------------------
	# The child has adopted the faith previously tracked.
	# The old exposure is no longer relevant.
	# ---------------------------------------------------------

	else_if = {
		limit = {
			exists = var:nu_belief_exposure_target
			var:nu_belief_exposure_target = root.faith
		}

		remove_variable = nu_belief_exposure_target
		remove_variable = nu_belief_exposure_years
		remove_variable = nu_belief_exposure_score
		remove_variable = nu_belief_exposure_floor
	}

	# ---------------------------------------------------------
	# No current foreign religious influence.
	# Previous exposure fades gradually.
	# ---------------------------------------------------------

	else_if = {
		limit = {
			has_variable = nu_belief_exposure_score
		}

		change_variable = {
			name = nu_belief_exposure_score
			add = -5
		}

		if = {
			limit = {
				has_variable = nu_belief_exposure_floor
				var:nu_belief_exposure_floor >= 50
				var:nu_belief_exposure_score < 50
			}

			set_variable = {
				name = nu_belief_exposure_score
				value = 50
			}
		}

		else_if = {
			limit = {
				has_variable = nu_belief_exposure_floor
				var:nu_belief_exposure_floor >= 30
				var:nu_belief_exposure_score < 30
			}

			set_variable = {
				name = nu_belief_exposure_score
				value = 30
			}
		}

		else_if = {
			limit = {
				var:nu_belief_exposure_score <= 0
			}

			remove_variable = nu_belief_exposure_target
			remove_variable = nu_belief_exposure_years
			remove_variable = nu_belief_exposure_score
			remove_variable = nu_belief_exposure_floor
		}
	}
}
