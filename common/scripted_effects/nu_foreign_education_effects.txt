# =============================================================
# NATURAL UPBRINGING - FOREIGN EDUCATION NEWS
# =============================================================
# Root: child.
# One ordinary education-news roll per eligible birthday, with a
# common two-year cooldown. Mechanical exposure changes are resolved
# before the parent receives the letter and are not exposed in UI.
# =============================================================


nu_check_foreign_education_news_effect = {

	if = {
		limit = {
			age >= 6
			is_adult = no
			has_education_focus_trigger = yes
			nu_is_genuinely_educated_abroad_trigger = yes

			any_parent = {
				is_alive = yes
				is_ai = no
			}

			NOT = { has_character_flag = nu_foreign_education_news_cooldown }
			NOT = { has_character_flag = nu_assimilation_happened_this_birthday }
			NOT = { has_character_flag = nu_homesickness_request_pending }
			NOT = { has_character_flag = nu_runaway_active }
		}

		random_relation = {
			type = guardian
			save_scope_as = nu_foreign_education_guardian
		}

		var:nu_guardianship_home_court_owner = {
			save_temporary_scope_as = nu_foreign_education_home_court
		}

		court_owner = {
			save_scope_as = nu_foreign_education_current_court
			culture = {
				save_scope_as = nu_foreign_education_current_culture
			}
			faith = {
				save_scope_as = nu_foreign_education_current_faith
			}
			rite = {
				save_scope_as = nu_foreign_education_current_rite
			}
		}

		if = {
			limit = {
				scope:nu_foreign_education_current_court != scope:nu_foreign_education_home_court
				scope:nu_foreign_education_guardian = {
					court_owner ?= {
						this = scope:nu_foreign_education_current_court
					}
				}
			}

			random = {
				chance = nu_foreign_education_news_chance

				add_character_flag = {
					flag = nu_foreign_education_news_cooldown
					years = 2
				}

				save_scope_as = nu_foreign_education_child

				random_list = {

					# 1. General adaptation report
					30 = {
						modifier = {
							factor = 3
							NOT = { has_character_flag = nu_foreign_education_first_report_seen }
						}

						add_character_flag = nu_foreign_education_first_report_seen
						nu_resolve_general_adaptation_report_effect = yes

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0001
						}
					}

					# 2. Unexpected friendship
					20 = {
						trigger = {
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									age >= 6
									is_adult = no
									this != root
									can_set_relation_friend_trigger = { CHARACTER = root }
								}
							}
						}

						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									age >= 6
									is_adult = no
									this != root
									can_set_relation_friend_trigger = { CHARACTER = root }
								}
								save_scope_as = nu_foreign_education_peer
							}
						}

						if = {
							limit = { exists = scope:nu_foreign_education_peer }
							set_relation_friend = {
								reason = friend_childhood_random
								target = scope:nu_foreign_education_peer
							}
							nu_resolve_unexpected_friendship_effect = yes
						}

						# A reportable crush created by this friendship uses
						# the dedicated crush letter. A secret crush does not.
						if = {
							limit = {
								has_character_flag = nu_foreign_crush_from_friendship
								has_character_flag = nu_foreign_crush_reportable
							}
							every_parent = {
								limit = { is_alive = yes is_ai = no }
								trigger_event = nu_foreign_education.0070
							}
						}
						else = {
							every_parent = {
								limit = { is_alive = yes is_ai = no }
								trigger_event = nu_foreign_education.0010
							}
						}
					}

					# 3. The foreigner at court
					18 = {
						trigger = {
							scope:nu_foreign_education_current_culture != root.culture
						}

						scope:nu_foreign_education_current_court = {
							if = {
								limit = {
									any_courtier_or_guest = {
										age >= 6
										is_adult = no
										this != root
									}
								}
								random_courtier_or_guest = {
									limit = {
										age >= 6
										is_adult = no
										this != root
									}
									save_scope_as = nu_foreign_education_peer
								}
							}
						}

						nu_resolve_foreigner_at_court_effect = yes

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0020
						}
					}

					# 4. A strange custom
					18 = {
						trigger = {
							OR = {
								scope:nu_foreign_education_current_culture != root.culture
								scope:nu_foreign_education_current_faith != root.faith
								scope:nu_foreign_education_current_rite != root.rite
							}
						}

						nu_resolve_strange_custom_effect = yes

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0030
						}
					}

					# 5. Familiar words
					12 = {
						trigger = {
							exists = var:nu_culture_exposure_target
							has_variable = nu_culture_exposure_score
							var:nu_culture_exposure_target = scope:nu_foreign_education_current_culture
							var:nu_culture_exposure_score >= 20
						}

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0040
						}
					}

					# 6. Between two worlds
					8 = {
						trigger = {
							NOT = { has_character_flag = nu_foreign_education_two_worlds_seen }
							exists = var:nu_culture_exposure_target
							has_variable = nu_culture_exposure_score
							var:nu_culture_exposure_target = scope:nu_foreign_education_current_culture
							var:nu_culture_exposure_score >= 30
							culture != scope:nu_foreign_education_current_culture
						}

						add_character_flag = nu_foreign_education_two_worlds_seen

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0050
						}
					}

					# 7. The crush
					10 = {
						trigger = {
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									nu_is_valid_foreign_crush_for_root_trigger = yes
									nu_is_reportable_foreign_crush_for_root_trigger = yes
								}
							}
						}

						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									nu_is_valid_foreign_crush_for_root_trigger = yes
									nu_is_reportable_foreign_crush_for_root_trigger = yes
								}
								save_scope_as = nu_foreign_education_peer
							}
						}

						nu_make_foreign_crush_effect = yes

						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_foreign_education.0070
						}
					}

					# 8. The rival
					10 = {
						trigger = {
							any_relation = {
								type = rival
								court_owner ?= {
									this = scope:nu_foreign_education_current_court
								}
							}
						}

						random_relation = {
							type = rival
							limit = {
								court_owner ?= {
									this = scope:nu_foreign_education_current_court
								}
							}
							save_scope_as = nu_foreign_education_rival
						}

						nu_resolve_rival_news_effect = yes

						# If a reportable crush emerged from the rivalry,
						# use the crush letter with the rivalry variant.
						if = {
							limit = {
								has_character_flag = nu_rival_news_crush
								has_character_flag = nu_foreign_crush_reportable
							}
							scope:nu_foreign_education_rival = {
								save_scope_as = nu_foreign_education_peer
							}
							every_parent = {
								limit = { is_alive = yes is_ai = no }
								trigger_event = nu_foreign_education.0070
							}
						}
						else = {
							every_parent = {
								limit = { is_alive = yes is_ai = no }
								trigger_event = nu_foreign_education.0080
							}


					# -------------------------------------------------
					# FOCUS-SPECIFIC EDUCATION NEWS
					# Two entries become eligible for the current focus.
					# -------------------------------------------------

					# Diplomacy 1: Between Two Quarrels
					15 = {
						trigger = {
							has_focus = education_diplomacy
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									age >= 6
									is_adult = no
									this != root
								}
							}
						}
						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									age >= 6
									is_adult = no
									this != root
								}
								save_scope_as = nu_focus_target
							}
						}
						nu_resolve_focus_diplomacy_quarrels_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0100
						}
					}

					# Diplomacy 2: A Face From Elsewhere
					15 = {
						trigger = {
							has_focus = education_diplomacy
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									this != root
									is_alive = yes
								}
							}
						}
						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									this != root
									is_alive = yes
								}
								save_scope_as = nu_focus_target
							}
						}
						nu_resolve_focus_diplomacy_visitor_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0110
						}
					}

					# Martial 1: The Field Rather Than the Books
					15 = {
						trigger = { has_focus = education_martial }
						nu_resolve_focus_martial_field_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0120
						}
					}

					# Martial 2: One Blow Too Many
					15 = {
						trigger = {
							has_focus = education_martial
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									age >= 6
									is_adult = no
									this != root
								}
							}
						}
						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									age >= 6
									is_adult = no
									this != root
								}
								save_scope_as = nu_focus_peer
							}
						}
						nu_resolve_focus_martial_blow_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0130
						}
					}

					# Stewardship 1: Something Does Not Add Up
					15 = {
						trigger = { has_focus = education_stewardship }
						nu_resolve_focus_stewardship_count_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0140
						}
					}

					# Stewardship 2: Making Supplies Last
					15 = {
						trigger = { has_focus = education_stewardship }
						nu_resolve_focus_stewardship_reserves_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0150
						}
					}

					# Intrigue 1: Not Meant for Their Ears
					15 = {
						trigger = {
							has_focus = education_intrigue
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									is_adult = yes
									this != root
								}
							}
						}
						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									is_adult = yes
									this != root
								}
								save_scope_as = nu_focus_target
							}
						}
						nu_resolve_focus_intrigue_overheard_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0160
						}
					}

					# Intrigue 2: I Know Something You Do Not
					15 = {
						trigger = {
							has_focus = education_intrigue
							scope:nu_foreign_education_current_court = {
								any_courtier_or_guest = {
									is_adult = yes
									this != root
								}
							}
						}
						scope:nu_foreign_education_current_court = {
							random_courtier_or_guest = {
								limit = {
									is_adult = yes
									this != root
								}
								save_scope_as = nu_focus_target
							}
						}
						nu_resolve_focus_intrigue_leverage_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0170
						}
					}

					# Learning 1: Two Teachings, Two Truths
					15 = {
						trigger = { has_focus = education_learning }
						nu_resolve_focus_learning_teachings_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0180
						}
					}

					# Learning 2: Why?
					15 = {
						trigger = { has_focus = education_learning }
						nu_resolve_focus_learning_why_effect = yes
						every_parent = {
							limit = { is_alive = yes is_ai = no }
							trigger_event = nu_focus_education.0190
						}
					}
						}
					}
				}
			}
		}
	}
}
