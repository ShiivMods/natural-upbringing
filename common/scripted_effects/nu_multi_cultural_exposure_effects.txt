# Natural Upbringing 1.2.1
# Multi-cultural exposure engine.
#
# Root for every effect in this file is the character whose exposure is tracked.
# Persistent storage uses eight bounded slots. The old 1.2 single-target variables
# are rebuilt as a compatibility view of the strongest CURRENT foreign influence.

# =============================================================
# SLOT HELPERS
# =============================================================

nu_touch_cultural_exposure_slot_effect = {
	# Required:
	# - scope:nu_culture_target = Culture
	# - var:nu_tmp_culture_gain = numeric gain for this source
	# - $SLOT$ = 1..8

	if = {
		limit = { NOT = { has_variable = nu_culture_exposure_score_$SLOT$ } }
		set_variable = { name = nu_culture_exposure_score_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = nu_culture_exposure_pressure_$SLOT$ } }
		set_variable = { name = nu_culture_exposure_pressure_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = nu_culture_exposure_years_$SLOT$ } }
		set_variable = { name = nu_culture_exposure_years_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = nu_culture_exposure_streak_$SLOT$ } }
		set_variable = { name = nu_culture_exposure_streak_$SLOT$ value = 0 }
	}

	# Years/streak advance only once per yearly update, even when several
	# independent sources reinforce the same culture.
	if = {
		limit = { NOT = { has_character_flag = nu_culture_exposure_touched_$SLOT$ } }
		add_character_flag = nu_culture_exposure_touched_$SLOT$
		change_variable = { name = nu_culture_exposure_years_$SLOT$ add = 1 }
		change_variable = { name = nu_culture_exposure_streak_$SLOT$ add = 1 }
	}

	change_variable = {
		name = nu_culture_exposure_score_$SLOT$
		add = var:nu_tmp_culture_gain
	}
	change_variable = {
		name = nu_culture_exposure_pressure_$SLOT$
		add = var:nu_tmp_culture_gain
	}

	if = {
		limit = { var:nu_culture_exposure_score_$SLOT$ > 100 }
		set_variable = { name = nu_culture_exposure_score_$SLOT$ value = 100 }
	}
	if = {
		limit = { var:nu_culture_exposure_score_$SLOT$ < 0 }
		set_variable = { name = nu_culture_exposure_score_$SLOT$ value = 0 }
	}
}

nu_apply_cultural_source_effect = {
	# Required:
	# - scope:nu_culture_target
	# - var:nu_tmp_culture_source_base

	set_variable = {
		name = nu_tmp_culture_gain
		value = var:nu_tmp_culture_source_base
	}

	# Cultural distance only accelerates FOREIGN integration.
	if = {
		limit = { scope:nu_culture_target != root.culture }

		set_variable = { name = nu_tmp_culture_proximity value = 1 }

		if = {
			limit = {
				culture = {
					has_same_culture_heritage = scope:nu_culture_target
				}
			}
			change_variable = { name = nu_tmp_culture_proximity multiply = 1.20 }
		}

		if = {
			limit = {
				culture = {
					has_same_culture_language = scope:nu_culture_target
				}
			}
			change_variable = { name = nu_tmp_culture_proximity multiply = 1.20 }
		}

		if = {
			limit = { knows_language_of_culture = scope:nu_culture_target }
			change_variable = { name = nu_tmp_culture_proximity multiply = 1.25 }
		}

		if = {
			limit = { var:nu_tmp_culture_proximity > 1.60 }
			set_variable = { name = nu_tmp_culture_proximity value = 1.60 }
		}

		change_variable = {
			name = nu_tmp_culture_gain
			multiply = var:nu_tmp_culture_proximity
		}
	}

	# Age: exposure starts at birth; adults continue learning but much more slowly.
	if = {
		limit = { age <= 5 }
		change_variable = { name = nu_tmp_culture_gain multiply = 1.50 }
	}
	else_if = {
		limit = { age <= 9 }
		change_variable = { name = nu_tmp_culture_gain multiply = 1.25 }
	}
	else_if = {
		limit = { age <= 12 }
		# x1.00
	}
	else_if = {
		limit = { age <= 15 }
		change_variable = { name = nu_tmp_culture_gain multiply = 0.75 }
	}
	else = {
		change_variable = { name = nu_tmp_culture_gain multiply = 0.50 }
	}

	# County synergy: local society strengthens matching household/social input.
	if = {
		limit = {
			NOT = { has_variable = nu_tmp_culture_direct_county_source }
			exists = scope:nu_local_culture
			scope:nu_local_culture = scope:nu_culture_target
		}
		change_variable = { name = nu_tmp_culture_gain multiply = 1.25 }
	}

	nu_add_cultural_influence_effect = yes

	remove_variable = nu_tmp_culture_gain
	remove_variable = nu_tmp_culture_proximity
}

nu_resolve_local_culture_effect = {
	# Prefer where the character physically is.
	if = {
		limit = { location ?= { exists = county } }
		location.county.culture = { save_temporary_scope_as = nu_local_culture }
	}

	# Landless owner fallback: the camp/domicile location.
	else_if = {
		limit = {
			is_landless_adventurer = yes
			domicile ?= {
				domicile_location ?= { exists = county }
			}
		}
		domicile.domicile_location.county.culture = {
			save_temporary_scope_as = nu_local_culture
		}
	}

	# Member of an adventurer camp.
	else_if = {
		limit = {
			court_owner ?= {
				is_landless_adventurer = yes
				domicile ?= {
					domicile_location ?= { exists = county }
				}
			}
		}
		court_owner.domicile.domicile_location.county.culture = {
			save_temporary_scope_as = nu_local_culture
		}
	}

	# Last-resort compatibility with the pre-1.2.1 court model.
	else_if = {
		limit = {
			court_owner ?= {
				capital_province ?= { exists = county }
			}
		}
		court_owner.capital_province.county.culture = {
			save_temporary_scope_as = nu_local_culture
		}
	}
}

nu_apply_social_cultural_sources_effect = {
	if = {
		limit = { exists = court_owner }

		court_owner = { save_temporary_scope_as = nu_social_owner }
		set_variable = { name = nu_tmp_social_total value = 0 }

		scope:nu_social_owner = {
			root = {
				change_variable = { name = nu_tmp_social_total add = 1 }
			}
			culture = {
				add_to_temporary_list = nu_social_cultures
			}

			every_courtier = {
				root = {
					change_variable = { name = nu_tmp_social_total add = 1 }
				}
				culture = {
					if = {
						limit = { NOT = { is_in_list = nu_social_cultures } }
						add_to_temporary_list = nu_social_cultures
					}
				}
			}
		}

		if = {
			limit = { var:nu_tmp_social_total > 0 }

			every_in_list = {
				list = nu_social_cultures
				save_temporary_scope_as = nu_social_test_culture

				root = {
					set_variable = { name = nu_tmp_social_match value = 0 }
				}

				scope:nu_social_owner = {
					if = {
						limit = { culture = scope:nu_social_test_culture }
						root = {
							change_variable = { name = nu_tmp_social_match add = 1 }
						}
					}
					every_courtier = {
						limit = { culture = scope:nu_social_test_culture }
						root = {
							change_variable = { name = nu_tmp_social_match add = 1 }
						}
					}
				}

				root = {
					# Dominant means at least 33% of Court/Camp members.
					if = {
						limit = {
							var:nu_tmp_social_match >= {
								value = var:nu_tmp_social_total
								multiply = 0.33
							}
						}

						scope:nu_social_test_culture = {
							save_temporary_scope_as = nu_culture_target
						}

						set_variable = {
							name = nu_tmp_culture_source_base
							value = var:nu_tmp_social_match
						}
						change_variable = { name = nu_tmp_culture_source_base multiply = 3 }
						change_variable = { name = nu_tmp_culture_source_base divide = var:nu_tmp_social_total }

						nu_apply_cultural_source_effect = yes
						remove_variable = nu_tmp_culture_source_base
					}
				}
			}
		}

		remove_variable = nu_tmp_social_total

nu_seed_birth_social_culture_effect = {
	# Exactly 25 birth-share points are distributed among dominant Court/Camp
	# cultures in proportion to their representation.
	if = {
		limit = { exists = court_owner }

		court_owner = { save_temporary_scope_as = nu_birth_social_owner }
		set_variable = { name = nu_tmp_birth_social_total value = 0 }
		set_variable = { name = nu_tmp_birth_dominant_total value = 0 }

		scope:nu_birth_social_owner = {
			root = { change_variable = { name = nu_tmp_birth_social_total add = 1 } }
			culture = { add_to_temporary_list = nu_birth_social_cultures }

			every_courtier = {
				root = { change_variable = { name = nu_tmp_birth_social_total add = 1 } }
				culture = {
					if = {
						limit = { NOT = { is_in_list = nu_birth_social_cultures } }
						add_to_temporary_list = nu_birth_social_cultures
					}
				}
			}
		}

		if = {
			limit = { var:nu_tmp_birth_social_total > 0 }

			every_in_list = {
				list = nu_birth_social_cultures
				save_temporary_scope_as = nu_birth_test_culture
				root = { set_variable = { name = nu_tmp_birth_match value = 0 } }

				scope:nu_birth_social_owner = {
					if = {
						limit = { culture = scope:nu_birth_test_culture }
						root = { change_variable = { name = nu_tmp_birth_match add = 1 } }
					}
					every_courtier = {
						limit = { culture = scope:nu_birth_test_culture }
						root = { change_variable = { name = nu_tmp_birth_match add = 1 } }
					}
				}

				root = {
					if = {
						limit = {
							var:nu_tmp_birth_match >= {
								value = var:nu_tmp_birth_social_total
								multiply = 0.33
							}
						}
						scope:nu_birth_test_culture = {
							add_to_temporary_list = nu_birth_dominant_cultures
						}
						change_variable = {
							name = nu_tmp_birth_dominant_total
							add = var:nu_tmp_birth_match
						}
					}
				}
			}
		}

		if = {
			limit = { var:nu_tmp_birth_dominant_total > 0 }

			every_in_list = {
				list = nu_birth_dominant_cultures
				save_temporary_scope_as = nu_birth_test_culture
				root = { set_variable = { name = nu_tmp_birth_match value = 0 } }

				scope:nu_birth_social_owner = {
					if = {
						limit = { culture = scope:nu_birth_test_culture }
						root = { change_variable = { name = nu_tmp_birth_match add = 1 } }
					}
					every_courtier = {
						limit = { culture = scope:nu_birth_test_culture }
						root = { change_variable = { name = nu_tmp_birth_match add = 1 } }
					}
				}

				root = {
					scope:nu_birth_test_culture = {
						save_temporary_scope_as = nu_culture_target
					}
					set_variable = { name = nu_tmp_culture_seed value = 25 }
					change_variable = { name = nu_tmp_culture_seed multiply = var:nu_tmp_birth_match }
					change_variable = { name = nu_tmp_culture_seed divide = var:nu_tmp_birth_dominant_total }
					nu_seed_cultural_exposure_effect = yes
					remove_variable = nu_tmp_culture_seed
				}
			}
		}
		else = {
			# No usable social context: protect the child's inherited identity.
			culture = { save_temporary_scope_as = nu_culture_target }
			set_variable = { name = nu_tmp_culture_seed value = 25 }
			nu_seed_cultural_exposure_effect = yes
			remove_variable = nu_tmp_culture_seed
		}

		remove_variable = nu_tmp_birth_social_total
		remove_variable = nu_tmp_birth_dominant_total
		remove_variable = nu_tmp_birth_match
	}
	else = {
		culture = { save_temporary_scope_as = nu_culture_target }
		set_variable = { name = nu_tmp_culture_seed value = 25 }
		nu_seed_cultural_exposure_effect = yes
		remove_variable = nu_tmp_culture_seed
	}
}

nu_initialize_cultural_exposure_at_birth_effect = {
	if = {
		limit = { NOT = { has_character_flag = nu_multi_culture_exposure_initialized } }

		culture = { save_temporary_scope_as = nu_culture_target }
		set_variable = { name = nu_tmp_culture_seed value = 25 }
		nu_seed_cultural_exposure_effect = yes

		# 25: mother, fallback to current Culture.
		if = {
			limit = { exists = scope:mother }
			scope:mother.culture = { save_temporary_scope_as = nu_culture_target }
		}
		else = {
			culture = { save_temporary_scope_as = nu_culture_target }
		}
		set_variable = { name = nu_tmp_culture_seed value = 25 }
		nu_seed_cultural_exposure_effect = yes

		# 25: family father, fallback to current Culture.
		if = {
			limit = { exists = scope:father }
			scope:father.culture = { save_temporary_scope_as = nu_culture_target }
		}
		else = {
			culture = { save_temporary_scope_as = nu_culture_target }
		}
		set_variable = { name = nu_tmp_culture_seed value = 25 }
		nu_seed_cultural_exposure_effect = yes

		# 25 total: immediate Court/Camp social environment.
		nu_seed_birth_social_culture_effect = yes

		remove_variable = nu_tmp_culture_seed
		add_character_flag = nu_multi_culture_exposure_initialized
		add_character_flag = nu_multi_culture_exposure_migrated
	}
}

nu_migrate_legacy_cultural_exposure_effect = {
	if = {
		limit = { NOT = { has_character_flag = nu_multi_culture_exposure_initialized } }

			}

			set_variable = { name = nu_culture_exposure_score_1 value = var:nu_culture_exposure_score }

				limit = { has_variable = nu_culture_exposure_years }
				set_variable = { name = nu_culture_exposure_years_1 value = var:nu_culture_exposure_years }
				set_variable = { name = nu_culture_exposure_streak_1 value = var:nu_culture_exposure_years }
			}
			else = {
				set_variable = { name = nu_culture_exposure_years_1 value = 0 }
				set_variable = { name = nu_culture_exposure_streak_1 value = 0 }
			}

			if = {
				limit = { var:nu_culture_exposure_score_1 > 100 }
				set_variable = { name = nu_culture_exposure_score_1 value = 100 }
			}

			# Existing characters need a rooted current identity too.
			if = {
				limit = { var:nu_culture_exposure_target_1 != root.culture }
				culture = { save_temporary_scope_as = nu_culture_target }
				set_variable = { name = nu_tmp_culture_seed value = 60 }
				nu_seed_cultural_exposure_effect = yes
			}
			else_if = {
				limit = { var:nu_culture_exposure_score_1 < 60 }
				set_variable = { name = nu_culture_exposure_score_1 value = 60 }
			}
		}
		else = {
			# No legacy history: start existing characters with a stable identity.
			culture = { save_temporary_scope_as = nu_culture_target }
			set_variable = { name = nu_tmp_culture_seed value = 60 }
			nu_seed_cultural_exposure_effect = yes
		}

		remove_variable = nu_tmp_culture_seed
		add_character_flag = nu_multi_culture_exposure_initialized
		add_character_flag = nu_multi_culture_exposure_migrated
	}

	# Legacy values are now a compatibility view rebuilt after each update.
	remove_variable = nu_culture_exposure_target
	remove_variable = nu_culture_exposure_score
	remove_variable = nu_culture_exposure_years
	remove_variable = nu_culture_exposure_floor
}

	remove_character_flag = nu_culture_exposure_touched_7
	remove_character_flag = nu_culture_exposure_touched_8

	# Active pressure is rebuilt from scratch each year. It measures the
	# strength of the CURRENT environment, not historical familiarity.
	if = {
		limit = { exists = var:nu_culture_exposure_target_1 }
		set_variable = { name = nu_culture_exposure_pressure_1 value = 0 }
	}
	else = {
		remove_variable = nu_culture_exposure_pressure_1
	}
	if = {
		limit = { exists = var:nu_culture_exposure_target_2 }

	}

	nu_resolve_local_culture_effect = yes

	# Parents: medium/strong direct influence.
	every_parent = {
		limit = { is_alive = yes }
		culture = { save_temporary_scope_as = nu_culture_target }
		root = {
			set_variable = { name = nu_tmp_culture_source_base value = 2.5 }
			nu_apply_cultural_source_effect = yes
			remove_variable = nu_tmp_culture_source_base
		}
	}

	# Guardian: strongest individual educational influence.
	if = {
		limit = { exists = scope:nu_guardian }
		scope:nu_guardian.culture = { save_temporary_scope_as = nu_culture_target }
		set_variable = { name = nu_tmp_culture_source_base value = 4 }
		nu_apply_cultural_source_effect = yes
		remove_variable = nu_tmp_culture_source_base
	}

	# Court/Camp demographic environment. Several cultures can qualify.
	nu_apply_social_cultural_sources_effect = yes

	# Liege: weak political/cultural influence. This is deliberately much
	# weaker than a parent, guardian or dominant household environment.
	if = {
		limit = { exists = liege }
		liege.culture = { save_temporary_scope_as = nu_culture_target }

		scope:nu_local_culture = { save_temporary_scope_as = nu_culture_target }

	}

	nu_decay_cultural_exposure_effect = yes

	remove_variable = nu_tmp_culture_source_base
	remove_variable = nu_tmp_culture_gain
	remove_variable = nu_tmp_culture_proximity
	remove_variable = nu_tmp_culture_direct_county_source
	remove_variable = nu_tmp_social_total
	remove_variable = nu_tmp_social_match
