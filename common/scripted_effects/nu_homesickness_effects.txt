nu_clear_homesickness_effect = {
	remove_variable = nu_homesickness_active_years
	remove_variable = nu_homesickness_severity
	remove_character_flag = nu_homesickness_active
	remove_character_flag = nu_homesickness_request_pending

	# Legacy 1.1 / early 1.2 state. Repeated requests are no longer used.
	remove_character_flag = nu_homesickness_cooldown
}


# Root: child.
# Starts one persistent homesickness state and sends its single request home.
nu_begin_homesickness_effect = {
	if = {
		limit = {
			NOT = { has_character_flag = nu_homesickness_active }
			is_adult = no
			exists = court_owner
			any_relation = {
				type = guardian
				always = yes
			}
			any_parent = {
				is_alive = yes
			}
		}

		add_character_flag = nu_homesickness_active
		set_variable = {
			name = nu_homesickness_active_years
			value = 0
		}
		set_variable = {
			name = nu_homesickness_severity
			value = 1
		}
		add_character_flag = nu_homesickness_request_pending

		save_scope_as = nu_homesick_child

		random_relation = {
			type = guardian
			save_scope_as = nu_homesick_guardian
		}

		court_owner = {
			save_scope_as = nu_homesick_current_court
		}

		# One request only for this continuous homesickness state.
		if = {
			limit = {
				any_parent = {
					is_alive = yes
					is_ai = no
				}
			}

			random_parent = {
				limit = {
					is_alive = yes
					is_ai = no
				}
				trigger_event = nu_homesickness.0001
			}
		}
		else = {
			random_parent = {
				limit = {
					is_alive = yes
				}
				trigger_event = nu_homesickness.0001
			}
		}
	}
}


nu_check_homesickness_effect = {

	# =========================================================
	# BASIC ELIGIBILITY
	# =========================================================

	if = {
		limit = {
			age >= 6
			is_adult = no
			exists = var:nu_guardianship_home_court_owner
			any_relation = {
				type = guardian
				always = yes
			}
			exists = court_owner
		}

		random_relation = {
			type = guardian
			save_temporary_scope_as = nu_homesickness_guardian
		}

		var:nu_guardianship_home_court_owner = {
			save_temporary_scope_as = nu_homesickness_home_court
		}

		court_owner = {
			save_temporary_scope_as = nu_homesickness_current_court

			culture = {
				save_temporary_scope_as = nu_homesickness_current_culture
			}

			faith = {
				save_temporary_scope_as = nu_homesickness_current_faith
			}

			rite = {
				save_temporary_scope_as = nu_homesickness_current_rite
			}
		}

		# =========================================================
		# ACTUALLY LIVING AWAY WITH THE GUARDIAN
		# =========================================================

		if = {
			limit = {
				scope:nu_homesickness_current_court != scope:nu_homesickness_home_court

				scope:nu_homesickness_guardian = {
					court_owner ?= {
						this = scope:nu_homesickness_current_court
					}
				}
			}

			# Track continuous time abroad in this guardianship.
			if = {
				limit = {
					NOT = { has_variable = nu_homesickness_abroad_years }
				}
				set_variable = {
					name = nu_homesickness_abroad_years
					value = 1
				}
			}
			else = {
				change_variable = {
					name = nu_homesickness_abroad_years
					add = 1
				}
			}

			# Once homesickness exists it never fades naturally.
			# Each birthday simply records another full year lived with it.
			if = {
				limit = {
					has_character_flag = nu_homesickness_active
				}

				if = {
					limit = {
						NOT = { has_variable = nu_homesickness_active_years }
					}
					set_variable = {
						name = nu_homesickness_active_years
						value = 0
					}
				}

				change_variable = {
					name = nu_homesickness_active_years
					add = 1
				}
			}

			# Initial onset only. There are no repeated requests after refusal.
			else_if = {
				limit = {
					var:nu_homesickness_abroad_years >= 2
					NOT = { has_character_flag = nu_assimilation_happened_this_birthday }
					any_parent = {
						is_alive = yes
					}
				}

				random = {
					chance = nu_homesickness_chance
					nu_begin_homesickness_effect = yes
				}
			}
		}

		# =========================================================
		# NO LONGER IN THE FOREIGN GUARDIANSHIP
		# =========================================================
		else = {
			remove_variable = nu_homesickness_abroad_years
			nu_clear_homesickness_effect = yes
		}
	}

	# If the tracked guardianship itself disappeared, clear any lingering state.
	else_if = {
		limit = {
			OR = {
				is_adult = yes
				NOT = { any_relation = { type = guardian always = yes } }
				NOT = { exists = var:nu_guardianship_home_court_owner }
			}
			has_character_flag = nu_homesickness_active
		}
		remove_variable = nu_homesickness_abroad_years
		nu_clear_homesickness_effect = yes
	}
}
