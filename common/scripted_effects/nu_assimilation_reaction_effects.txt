# =============================================================
# NATURAL UPBRINGING 1.2 - ASSIMILATION REACTION SYSTEM
# =============================================================
#
# Reaction data is stored in local variables so each event keeps
# its own state. This prevents queued assimilation events, multiple
# children, or multiplayer parents from overwriting one another.
#
# Cultural and religious scores are calculated independently.
# Ties inside one domain keep the earlier family in this order:
# rejection > mistrust > acceptance > reflection > pragmatism > adaptation
# =============================================================

nu_select_cultural_assimilation_reaction_effect = {
	if = {
		limit = { local_var:nu_cultural_reaction = flag:rejection }
		set_local_variable = { name = nu_assimilation_reaction value = flag:rejection }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:mistrust }
		set_local_variable = { name = nu_assimilation_reaction value = flag:mistrust }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:acceptance }
		set_local_variable = { name = nu_assimilation_reaction value = flag:acceptance }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:reflection }
		set_local_variable = { name = nu_assimilation_reaction value = flag:reflection }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:pragmatism }
		set_local_variable = { name = nu_assimilation_reaction value = flag:pragmatism }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:adaptation }
		set_local_variable = { name = nu_assimilation_reaction value = flag:adaptation }
	}
	else_if = {
		limit = { local_var:nu_cultural_reaction = flag:fallback }
		set_local_variable = { name = nu_assimilation_reaction value = flag:fallback }
	}
}

nu_combine_assimilation_reactions_effect = {
	if = {
		limit = {
			local_var:nu_cultural_reaction = flag:rejection
			local_var:nu_religious_reaction = flag:rejection
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:rejection }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:mistrust
			local_var:nu_religious_reaction = flag:mistrust
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:mistrust }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:acceptance
			local_var:nu_religious_reaction = flag:acceptance
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:acceptance }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:reflection
			local_var:nu_religious_reaction = flag:reflection
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:reflection }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:pragmatism
			local_var:nu_religious_reaction = flag:pragmatism
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:pragmatism }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:adaptation
			local_var:nu_religious_reaction = flag:adaptation
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:adaptation }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
		set_local_variable = { name = nu_combined_reaction_reinforced value = yes }
	}
	else_if = {
		limit = {
			local_var:nu_cultural_reaction = flag:fallback
			local_var:nu_religious_reaction = flag:fallback
		}
		set_local_variable = { name = nu_assimilation_reaction value = flag:fallback }
		set_local_variable = { name = nu_combined_reaction_source value = flag:both }
	}
	else = {
		set_local_variable = { name = nu_reaction_compare value = local_var:nu_religious_reaction_strength }
		change_local_variable = { name = nu_reaction_compare subtract = local_var:nu_cultural_reaction_strength }

		if = {
			limit = { local_var:nu_reaction_compare > 0 }
			nu_select_religious_assimilation_reaction_effect = yes
			set_local_variable = { name = nu_combined_reaction_source value = flag:religious }
		}
		else_if = {
			limit = { local_var:nu_reaction_compare < 0 }
			nu_select_cultural_assimilation_reaction_effect = yes
			set_local_variable = { name = nu_combined_reaction_source value = flag:cultural }
		}
		else_if = {
			limit = {
				OR = {
					has_trait = zealous
					has_trait = cynical
				}
			}
			nu_select_religious_assimilation_reaction_effect = yes
			set_local_variable = { name = nu_combined_reaction_source value = flag:religious }
		}
		else = {
			nu_select_cultural_assimilation_reaction_effect = yes
			set_local_variable = { name = nu_combined_reaction_source value = flag:cultural }
		}
	}
}

nu_apply_assimilation_rejection_consequence_effect = {
	if = {
		limit = { NOT = { has_trait = callous } }

		if = {
			limit = {
				OR = {
					AND = {
						has_trait = zealous
						OR = {
							local_var:nu_assimilation_event_kind = flag:religion
							AND = {
								local_var:nu_assimilation_event_kind = flag:combined
								OR = {
									local_var:nu_combined_reaction_source = flag:religious
									local_var:nu_combined_reaction_source = flag:both
								}
							}
						}
					}
					exists = local_var:nu_combined_reaction_reinforced
				}
			}
			add_stress = medium_stress_gain
		}
		else = {
			add_stress = minor_stress_gain
		}
	}

	scope:nu_assimilated_child = {
		if = {
			limit = {
				root = {
					OR = {
						has_trait = sadistic
						AND = {
							has_trait = zealous
							OR = {
								local_var:nu_assimilation_event_kind = flag:religion
								AND = {
									local_var:nu_assimilation_event_kind = flag:combined
									OR = {
										local_var:nu_combined_reaction_source = flag:religious
										local_var:nu_combined_reaction_source = flag:both
									}
								}
							}
						}
					}
				}
			}
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -20
			}
		}
		else_if = {
			limit = { exists = local_var:nu_combined_reaction_reinforced }
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -15
			}
		}
		else = {
			add_opinion = {
				target = root
				modifier = disappointed_opinion
				opinion = -10
			}
		}
	}
}

nu_apply_assimilation_mistrust_consequence_effect = {
	if = {
		limit = { NOT = { has_trait = callous } }

		if = {
			limit = { exists = local_var:nu_combined_reaction_reinforced }
			if = {
				limit = { has_trait = brave }
				add_stress = minor_stress_gain
			}
			else = {
				add_stress = medium_stress_gain
			}
		}
		else_if = {
			limit = { NOT = { has_trait = brave } }
			add_stress = minor_stress_gain
		}
	}
}
