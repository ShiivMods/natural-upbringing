# Natural Upbringing 1.2.1
# Simultaneous Faith/Rite exposure engine.
#
# Faith always has priority over Rite:
# - different Faith -> Faith exposure only;
# - same Faith + different Rite -> Rite exposure only.

# SLOT HELPERS
# =============================================================

nu_touch_belief_exposure_slot_effect = {
	if = {
		limit = { NOT = { has_variable = nu_belief_exposure_score_$SLOT$ } }
		set_variable = { name = nu_belief_exposure_score_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = nu_belief_exposure_years_$SLOT$ } }
		set_variable = { name = nu_belief_exposure_years_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_variable = nu_belief_exposure_streak_$SLOT$ } }
		set_variable = { name = nu_belief_exposure_streak_$SLOT$ value = 0 }
	}
	if = {
		limit = { NOT = { has_character_flag = nu_belief_exposure_touched_$SLOT$ } }
		add_character_flag = nu_belief_exposure_touched_$SLOT$
		change_variable = { name = nu_belief_exposure_years_$SLOT$ add = 1 }
		change_variable = { name = nu_belief_exposure_streak_$SLOT$ add = 1 }
	}
	change_variable = { name = nu_belief_exposure_score_$SLOT$ add = var:nu_tmp_belief_gain }
	if = {
		limit = { var:nu_belief_exposure_score_$SLOT$ > 100 }
		set_variable = { name = nu_belief_exposure_score_$SLOT$ value = 100 }
	}
}

nu_apply_belief_source_effect = {
	set_variable = { name = nu_tmp_belief_gain value = var:nu_tmp_religious_source_base }

	if = {
		limit = { age <= 5 }
		change_variable = { name = nu_tmp_belief_gain multiply = 1.50 }
	}
	else_if = {
		limit = { age <= 9 }
		change_variable = { name = nu_tmp_belief_gain multiply = 1.25 }
	}
	else_if = {
		limit = { age <= 12 }
	}
	else_if = {
		limit = { age <= 15 }
		change_variable = { name = nu_tmp_belief_gain multiply = 0.75 }
	}
	else = {
		change_variable = { name = nu_tmp_belief_gain multiply = 0.50 }
	}

	nu_add_belief_influence_effect = yes
	remove_variable = nu_tmp_belief_gain
}

nu_apply_religious_identity_source_effect = {
	# Required:
	# - scope:nu_religious_source_faith
	# - scope:nu_religious_source_rite
	# - var:nu_tmp_religious_source_base

	if = {
		limit = {
			exists = scope:nu_religious_source_faith
			scope:nu_religious_source_faith != root.faith
		}
		scope:nu_religious_source_faith = { save_temporary_scope_as = nu_belief_target }
		nu_apply_belief_source_effect = yes
	}
	else_if = {
		limit = {
			exists = scope:nu_religious_source_faith
			scope:nu_religious_source_faith = root.faith
			exists = scope:nu_religious_source_rite
			scope:nu_religious_source_rite != root.rite
		}
		scope:nu_religious_source_rite = { save_temporary_scope_as = nu_rite_target }
		nu_apply_rite_source_effect = yes
	}
}

nu_resolve_local_religion_effect = {
	if = {
		limit = { location ?= { exists = faith exists = rite } }
		location.faith = { save_temporary_scope_as = nu_local_faith }
		location.rite = { save_temporary_scope_as = nu_local_rite }
	}
	else_if = {
		limit = {
			is_landless_adventurer = yes
			domicile ?= { domicile_location ?= { exists = faith exists = rite } }
		}
		domicile.domicile_location.faith = { save_temporary_scope_as = nu_local_faith }
		domicile.domicile_location.rite = { save_temporary_scope_as = nu_local_rite }
	}
	else_if = {
		limit = {
			court_owner ?= {
				is_landless_adventurer = yes
				domicile ?= { domicile_location ?= { exists = faith exists = rite } }
			}
		}
		court_owner.domicile.domicile_location.faith = { save_temporary_scope_as = nu_local_faith }
		court_owner.domicile.domicile_location.rite = { save_temporary_scope_as = nu_local_rite }
	}
	else_if = {
		limit = {
			court_owner ?= { capital_province ?= { exists = faith exists = rite } }
		}
		court_owner.capital_province.faith = { save_temporary_scope_as = nu_local_faith }
		court_owner.capital_province.rite = { save_temporary_scope_as = nu_local_rite }
	}
}

nu_apply_social_religious_sources_effect = {
	if = {
		limit = { exists = court_owner }
		court_owner = { save_temporary_scope_as = nu_religious_social_owner }

		set_variable = { name = nu_tmp_religious_social_total value = 0 }
		set_variable = { name = nu_tmp_root_faith_social_total value = 0 }

		scope:nu_religious_social_owner = {
			root = { change_variable = { name = nu_tmp_religious_social_total add = 1 } }
			faith = {
				add_to_temporary_list = nu_social_faiths
			}
			if = {
				limit = { faith = root.faith }
				root = { change_variable = { name = nu_tmp_root_faith_social_total add = 1 } }
				rite = { add_to_temporary_list = nu_social_root_faith_rites }
			}

			every_courtier = {
				root = { change_variable = { name = nu_tmp_religious_social_total add = 1 } }
				faith = {
					if = {
						limit = { NOT = { is_in_list = nu_social_faiths } }
						add_to_temporary_list = nu_social_faiths
					}
				}
				if = {
					limit = { faith = root.faith }
					root = { change_variable = { name = nu_tmp_root_faith_social_total add = 1 } }
					rite = {
						if = {
							limit = { NOT = { is_in_list = nu_social_root_faith_rites } }
							add_to_temporary_list = nu_social_root_faith_rites
						}
					}
				}
			}
		}

		# Faith dominance uses the whole Court/Camp as denominator.
		if = {
			limit = { var:nu_tmp_religious_social_total > 0 }
			every_in_list = {
				list = nu_social_faiths
				save_temporary_scope_as = nu_social_test_faith
				root = { set_variable = { name = nu_tmp_social_faith_match value = 0 } }

				scope:nu_religious_social_owner = {
					if = {
						limit = { faith = scope:nu_social_test_faith }
						root = { change_variable = { name = nu_tmp_social_faith_match add = 1 } }
					}
					every_courtier = {
						limit = { faith = scope:nu_social_test_faith }
						root = { change_variable = { name = nu_tmp_social_faith_match add = 1 } }
					}
				}

				root = {
					if = {
						limit = {
							scope:nu_social_test_faith != root.faith
							var:nu_tmp_social_faith_match >= {
								value = var:nu_tmp_religious_social_total
								multiply = 0.33
							}
						}
						scope:nu_social_test_faith = { save_temporary_scope_as = nu_belief_target }
						set_variable = { name = nu_tmp_religious_source_base value = var:nu_tmp_social_faith_match }
						change_variable = { name = nu_tmp_religious_source_base multiply = 3 }
						change_variable = { name = nu_tmp_religious_source_base divide = var:nu_tmp_religious_social_total }
						nu_apply_belief_source_effect = yes
						remove_variable = nu_tmp_religious_source_base
					}
				}
			}
		}

		# Rite dominance is evaluated only inside the character's current Faith.
		if = {
			limit = { var:nu_tmp_root_faith_social_total > 0 }
			every_in_list = {
				list = nu_social_root_faith_rites
				save_temporary_scope_as = nu_social_test_rite
				root = { set_variable = { name = nu_tmp_social_rite_match value = 0 } }

				scope:nu_religious_social_owner = {
					if = {
						limit = {
							faith = root.faith
							rite = scope:nu_social_test_rite
						}
						root = { change_variable = { name = nu_tmp_social_rite_match add = 1 } }
					}
					every_courtier = {
						limit = {
							faith = root.faith
							rite = scope:nu_social_test_rite
						}
						root = { change_variable = { name = nu_tmp_social_rite_match add = 1 } }
					}
				}

				root = {
					if = {
						limit = {
							scope:nu_social_test_rite != root.rite
							var:nu_tmp_social_rite_match >= {
								value = var:nu_tmp_root_faith_social_total
								multiply = 0.33
							}
						}
						scope:nu_social_test_rite = { save_temporary_scope_as = nu_rite_target }
						set_variable = { name = nu_tmp_religious_source_base value = var:nu_tmp_social_rite_match }
						change_variable = { name = nu_tmp_religious_source_base multiply = 3 }
						change_variable = { name = nu_tmp_religious_source_base divide = var:nu_tmp_root_faith_social_total }
						nu_apply_rite_source_effect = yes
						remove_variable = nu_tmp_religious_source_base
					}
				}
			}
		}

		remove_variable = nu_tmp_religious_social_total
		remove_variable = nu_tmp_root_faith_social_total
		remove_variable = nu_tmp_social_faith_match
		remove_variable = nu_tmp_social_rite_match
	}
}
