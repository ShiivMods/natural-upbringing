# =============================================================
# NATURAL UPBRINGING - LIGHT PARENT REACTIONS TO EDUCATION NEWS
# =============================================================

nu_prepare_parent_news_reaction_effect = {
	set_local_variable = { name = nu_news_rejection value = 0 }
	set_local_variable = { name = nu_news_mistrust value = 0 }
	set_local_variable = { name = nu_news_acceptance value = 0 }
	set_local_variable = { name = nu_news_reflection value = 0 }
	set_local_variable = { name = nu_news_pragmatism value = 0 }
	set_local_variable = { name = nu_news_adaptation value = 0 }

	if = { limit = { has_trait = wrathful } change_local_variable = { name = nu_news_rejection add = 2 } }
	if = { limit = { has_trait = stubborn } change_local_variable = { name = nu_news_rejection add = 3 } }
	if = { limit = { has_trait = vengeful } change_local_variable = { name = nu_news_rejection add = 2 } }
	if = { limit = { has_trait = arrogant } change_local_variable = { name = nu_news_rejection add = 2 } }

	if = { limit = { has_trait = paranoid } change_local_variable = { name = nu_news_mistrust add = 3 } }
	if = { limit = { has_trait = craven } change_local_variable = { name = nu_news_mistrust add = 2 } }
	if = { limit = { has_trait = shy } change_local_variable = { name = nu_news_mistrust add = 1 } }

	if = { limit = { has_trait = compassionate } change_local_variable = { name = nu_news_acceptance add = 3 } }
	if = { limit = { has_trait = forgiving } change_local_variable = { name = nu_news_acceptance add = 2 } }
	if = { limit = { has_trait = humble } change_local_variable = { name = nu_news_acceptance add = 2 } }
	if = { limit = { has_trait = trusting } change_local_variable = { name = nu_news_acceptance add = 2 } }
	if = { limit = { has_trait = generous } change_local_variable = { name = nu_news_acceptance add = 1 } }

	if = { limit = { has_trait = calm } change_local_variable = { name = nu_news_reflection add = 2 } }
	if = { limit = { has_trait = patient } change_local_variable = { name = nu_news_reflection add = 2 } }
	if = { limit = { has_trait = temperate } change_local_variable = { name = nu_news_reflection add = 1 } }
	if = { limit = { has_trait = just } change_local_variable = { name = nu_news_reflection add = 2 } }
	if = { limit = { has_trait = honest } change_local_variable = { name = nu_news_reflection add = 1 } }
	if = { limit = { has_trait = diligent } change_local_variable = { name = nu_news_reflection add = 1 } }

	if = { limit = { has_trait = ambitious } change_local_variable = { name = nu_news_pragmatism add = 3 } }
	if = { limit = { has_trait = greedy } change_local_variable = { name = nu_news_pragmatism add = 2 } }
	if = { limit = { has_trait = deceitful } change_local_variable = { name = nu_news_pragmatism add = 2 } }
	if = { limit = { has_trait = cynical } change_local_variable = { name = nu_news_pragmatism add = 2 } }
	if = { limit = { has_trait = arbitrary } change_local_variable = { name = nu_news_pragmatism add = 1 } }

	if = { limit = { has_trait = content } change_local_variable = { name = nu_news_adaptation add = 2 } }
	if = { limit = { has_trait = fickle } change_local_variable = { name = nu_news_adaptation add = 3 } }
	if = { limit = { has_trait = lazy } change_local_variable = { name = nu_news_adaptation add = 1 } }
	if = { limit = { has_trait = gregarious } change_local_variable = { name = nu_news_adaptation add = 2 } }
	if = { limit = { has_trait = eccentric } change_local_variable = { name = nu_news_adaptation add = 2 } }

	set_local_variable = { name = nu_parent_news_reaction value = flag:rejection }
	set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_rejection }

	set_local_variable = { name = nu_news_compare value = local_var:nu_news_mistrust }
	change_local_variable = { name = nu_news_compare subtract = local_var:nu_parent_news_strength }
	if = {
		limit = { local_var:nu_news_compare > 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:mistrust }
		set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_mistrust }
	}

	set_local_variable = { name = nu_news_compare value = local_var:nu_news_acceptance }
	change_local_variable = { name = nu_news_compare subtract = local_var:nu_parent_news_strength }
	if = {
		limit = { local_var:nu_news_compare > 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:acceptance }
		set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_acceptance }
	}

	set_local_variable = { name = nu_news_compare value = local_var:nu_news_reflection }
	change_local_variable = { name = nu_news_compare subtract = local_var:nu_parent_news_strength }
	if = {
		limit = { local_var:nu_news_compare > 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:reflection }
		set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_reflection }
	}

	set_local_variable = { name = nu_news_compare value = local_var:nu_news_pragmatism }
	change_local_variable = { name = nu_news_compare subtract = local_var:nu_parent_news_strength }
	if = {
		limit = { local_var:nu_news_compare > 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:pragmatism }
		set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_pragmatism }
	}

	set_local_variable = { name = nu_news_compare value = local_var:nu_news_adaptation }
	change_local_variable = { name = nu_news_compare subtract = local_var:nu_parent_news_strength }
	if = {
		limit = { local_var:nu_news_compare > 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:adaptation }
		set_local_variable = { name = nu_parent_news_strength value = local_var:nu_news_adaptation }
	}

	if = {
		limit = { local_var:nu_parent_news_strength <= 0 }
		set_local_variable = { name = nu_parent_news_reaction value = flag:fallback }
	}
}


# Ordinary education news should not become a passive stress engine.
# Strong emotional consequences belong to the event itself when justified.
nu_apply_parent_news_rejection_effect = {}
nu_apply_parent_news_mistrust_effect = {}
nu_apply_parent_news_acceptance_effect = {}
