		var:nu_culture_exposure_target = {
			save_temporary_scope_as = nu_event_culture_target

		limit = {
			exists = var:nu_culture_exposure_target_1

			name = nu_culture_exposure_score_1

		}

			limit = { var:nu_culture_exposure_score_1 < 0 }

nu_change_current_religious_exposure_effect = {

	# corresponding to the current educational environment. They do not count
	# as an additional full year of exposure.

	if = {
		limit = {
			any_relation = {
				type = guardian
				always = yes
			}
		}
		random_relation = {
			type = guardian
			save_temporary_scope_as = nu_event_exposure_guardian
		}
	}

	if = {
		limit = {
			exists = scope:nu_event_exposure_guardian
			scope:nu_event_exposure_guardian.faith != root.faith
		}
		scope:nu_event_exposure_guardian.faith = {
			save_temporary_scope_as = nu_event_belief_environment
		}
	}
	else_if = {
		limit = {
			exists = scope:nu_event_exposure_guardian
			scope:nu_event_exposure_guardian.faith = root.faith
			scope:nu_event_exposure_guardian.rite != root.rite
		}
		scope:nu_event_exposure_guardian.rite = {
			save_temporary_scope_as = nu_event_rite_environment
		}
	}
	else_if = {
		limit = { court_owner ?= { faith != root.faith } }
		court_owner.faith = {
			save_temporary_scope_as = nu_event_belief_environment
		}
	}
	else_if = {
		limit = {
			court_owner ?= {
				faith = root.faith
				rite != root.rite
			}
		}
		court_owner.rite = {
			save_temporary_scope_as = nu_event_rite_environment
		}
	}

	set_variable = { name = nu_tmp_event_religious_delta value = $VALUE$ }
