nu_is_genuinely_educated_abroad_trigger = {

	age >= 6
	is_adult = no

	exists = var:nu_guardianship_home_court_owner
	exists = court_owner

	any_relation = {
		type = guardian
		always = yes
	}

	# The child must currently live away from the court recorded
	# when the guardianship began.
	trigger_if = {
		limit = {
			exists = var:nu_guardianship_home_court_owner
			exists = court_owner
		}

		NOT = {
			var:nu_guardianship_home_court_owner = {
				this = root.court_owner
			}
		}
	}

	# And the current court must genuinely be the guardian's court.
	trigger_if = {
		limit = {
			exists = court_owner
		}

		any_relation = {
			type = guardian

			court_owner ?= {
				this = root.court_owner
			}
		}
	}
}
