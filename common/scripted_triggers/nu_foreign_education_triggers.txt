nu_is_genuinely_educated_abroad_trigger = {

	age >= 6
	is_adult = no

	exists = var:nu_guardianship_home_court_owner
	exists = court_owner

	any_relation = {
		type = guardian
		always = yes
	}

	# The child must currently live away from the court recorded
	# when the guardianship began.
	trigger_if = {
		limit = {
			exists = var:nu_guardianship_home_court_owner
			exists = court_owner
		}

		NOT = {
			var:nu_guardianship_home_court_owner = {
				this = root.court_owner
			}
		}
	}

	# And the current court must genuinely be the guardian's court.
	trigger_if = {
		limit = {
			exists = court_owner
		}

		any_relation = {
			type = guardian

			court_owner ?= {
				this = root.court_owner
			}
		}
	}
}


# Root: child. CHARACTER: potential crush.
nu_can_form_foreign_crush_with_trigger = {
	is_adult = no
	age > childhood_education_start_age
	OR = {
		has_sexuality = heterosexual
		has_sexuality = bisexual
		has_sexuality = homosexual
	}
	is_attracted_to_gender_of = $CHARACTER$
	NOT = { has_relation_crush = $CHARACTER$ }

	$CHARACTER$ = {
		is_adult = no
		age > childhood_education_start_age
		age >= root.age_dif_child_down
		age <= root.age_dif_child_up
		this != root
		is_imprisoned = no
		is_close_family_of_root_trigger = no
	}
}


# Root: child. CHARACTER: crush.
# Different-sex crushes can be reported normally. Same-sex crushes
# remain private unless both children's rites accept homosexuality.
nu_foreign_crush_is_reportable_trigger = {
	OR = {
		sex_opposite_of = $CHARACTER$
		AND = {
			sex_same_as = $CHARACTER$
			rite = {
				rite_has_parameter = homosexuality_accepted
			}
			$CHARACTER$ = {
				rite = {
					rite_has_parameter = homosexuality_accepted
				}
			}
		}
	}
}


# Current scope: candidate. Root: child.
nu_is_valid_foreign_crush_for_root_trigger = {
	is_adult = no
	age > childhood_education_start_age
	age >= root.age_dif_child_down
	age <= root.age_dif_child_up
	this != root
	is_imprisoned = no
	is_close_family_of_root_trigger = no

	root = {
		is_adult = no
		age > childhood_education_start_age
		OR = {
			has_sexuality = heterosexual
			has_sexuality = bisexual
			has_sexuality = homosexual
		}
		is_attracted_to_gender_of = prev
		NOT = { has_relation_crush = prev }
	}
}


# Current scope: candidate. Root: child.
nu_is_reportable_foreign_crush_for_root_trigger = {
	OR = {
		sex_opposite_of = root
		AND = {
			sex_same_as = root
			rite = {
				rite_has_parameter = homosexuality_accepted
			}
			root.rite = {
				rite_has_parameter = homosexuality_accepted
			}
		}
	}
}
