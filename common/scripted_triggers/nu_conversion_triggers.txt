nu_has_cultural_assimilation_environment_trigger = {
	exists = scope:nu_conversion_culture

	# 1.2.1: validity is calculated from the persistent multi-track data.
	# It implies a current source, minimum score/streak, a clear lead over the
	# current Culture and competitors, and no active culture-change cooldown.
	has_character_flag = nu_cultural_assimilation_candidate_valid
}


nu_has_religious_assimilation_environment_trigger = {

	has_character_flag = nu_rite_assimilation_candidate_valid
}


nu_has_belief_assimilation_environment_trigger = {
	exists = scope:nu_conversion_belief
	has_character_flag = nu_belief_assimilation_candidate_valid
}


nu_has_active_religious_contract_protection_trigger = {

	exists = var:nu_religious_contract_protection_guardian

	# The protection remains active only while the guardian
	# attached to the protected arrangement is still the
	# child's actual guardian.
	any_relation = {
		type = guardian
		this = root.var:nu_religious_contract_protection_guardian
	}
}
